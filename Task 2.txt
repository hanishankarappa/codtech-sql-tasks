#Using a CTE(Common Table Expression) to Find Average Score per Student
mysql> WITH avg_scores AS (
    -> select student_id,name,AVG(score) AS avg_score
    -> FROM student_scores
    -> GROUP BY student_id,name)
    -> select * from avg_scores;

| student_id | name   | avg_score |

#Subquery to Find Students Above Class Average
mysql> select * from student_scores s
    -> where score >
    -> (
    -> select AVG(score)
    -> FROM student_scores
    -> WHERE class=s.class
    -> );

#Window Function (RANK) to Rank Students by Subject
mysql> SELECT
    -> name,
    -> subject,
    -> score,
    -> RANK() OVER(PARTITION BY subject ORDER BY score DESC) AS subject_rank
    -> FROM student_scores;

#Combined CTE + Window Function for Top Performers
mysql> WITH ranked_students AS(
    -> SELECT
    -> name,
    -> class,
    -> subject,
    -> score,
    -> DENSE_RANK() OVER (PARTITION BY class ORDER BY score DESC) AS rank_in_class
    -> FROM student_scores
    -> )
    -> select *
    -> FROM ranked_students
    -> WHERE rank_in_class<=2;
